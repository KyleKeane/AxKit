# Workflow: Find out whether a key arrives and what it does, in axClaude
# axClaude: Help, Diagnostics, Test a key (docs/axKit/SPEC.md AX-8.12; docs/axClaude/nvda-test-plan.md section 23). The
# user's own case of 2026-11-15: a function key that never reached the window. Every key is said with what it does
# where the focus is, the message field and then the conversation, and none is carried out: nothing typed, nothing
# sent to Claude, no notice; Escape twice ends it. Needs no Claude Code, no network and no model: a stand-in batch
# file draws one screen and waits, as in 22-background-work.
file home\AxKit\axClaude.json {"checkForUpdates":false}
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"}]}
mkdir kt
file claude.cmd @echo off\nif "%~1"=="--version" goto version\nchcp 65001 >nul\nfor /f %%a in ('echo prompt $E^| cmd') do set "E=%%a"\necho %E%[?25l%E%[2J%E%[Hclaude: started\necho auto mode on (shift+tab to cycle)\n<nul set /p "=$%E%[?25h"\n:idle\nping -n 2 127.0.0.1 >nul\ngoto idle\n:version\necho 0.0.0 (stand-in)

start claude {temp}\kt --claude {temp}\claude.cmd -- --model haiku
expect said Ready
expect commands "Test a key, Help, Diagnostics"
keys ctrl+1
expect focus Message
type hello
menu Test a key
expect said ! Key test. Press any key to hear its name and what it does here. Press Escape twice to end.

# Bundle keys and window keys, by their commands' names.
keys f8
expect said ! F8: Say the status
keys f4
expect said ! F4: Send a report
keys ctrl+b
expect said ! Ctrl+B: What runs in the background
keys ctrl+2
expect said ! Ctrl+2: Conversation
keys shift+escape
expect said ! Shift+Escape: Interrupt
# The message field's own keys: a letter is said and not typed; Enter is said and nothing is sent.
keys a
expect said ! A: types the letter
keys enter
expect said ! Enter: sends what the field holds
keys f9
expect said ! F9: no command of this window
expect state "notice":null
expect focus "value":"hello"

keys escape
expect said Press Escape again to end the key test
keys escape
expect said ! Key test ended
expect focus "value":"hello"

# The same keys act again: F8 says the status, Ctrl+2 goes to the conversation.
keys f8
expect said ! Claude: ready
keys ctrl+2
expect focus Conversation

# In the conversation its quick keys are named, a letter the view moves by and Ctrl+Alt with it; nothing moves.
menu Test a key
expect said ! Key test. Press any key
keys h
expect said ! H: moves to the next heading
keys shift+h
expect said ! Shift+H: moves to the previous heading
keys ctrl+alt+h
expect said ! Ctrl+Alt+H: reads every heading from the caret's line to the end
keys l
expect said ! L: says the line number
keys y
expect said ! Y: no command of this window
keys escape escape
expect said ! Key test ended
keys l
expect said ! Line
