# Workflow: Read back in the conversation while Claude writes on, and find every line in its place
# axClaude, docs/axClaude/nvda-test-plan.md section 24 (SPEC.md FR-3.3, FR-3.3a, FR-3.11): the conversation view holds
# its output back while text is selected in it, then takes everything that came meanwhile at once; a message sent
# with the caret left far up in the conversation, and one sent from a quoted line, go to the end of the conversation,
# never to the caret. The view checks its own text against the transcript's with every change and as it takes the
# focus, and a difference is an ERROR line in the log, which fails this run. Needs no Claude Code, no network and no
# model: the stand-in of Claude Code (tools/AxAutomate/StandIn.cs) draws one frame for each file the script makes in
# the project folder, and draws a typed message on its prompt row as Claude Code does, which a message's Enter waits
# for (D33; a batch file that drew nothing of it stood here until the fourth review of 2026-11-27).
file home\AxKit\axClaude.json {"replySpeech":"none","interruptForQuestions":true,"workingMusic":false,"backgroundAlert":false}
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"}]}
mkdir order
file claude.cmd @"%AXKIT_STANDIN%" standin "%~dp0scenario.json" %*
file scenario.json {"version":"0.0.0 (stand-in)","steps":[{"frame":["claude: started","auto mode on (shift+tab to cycle)","$"],"clear":true},{"waitFile":"go1"},{"frame":["claude: started","claude: one","claude: two","claude: three","auto mode on (shift+tab to cycle)","$"],"clear":true},{"waitFile":"go2"},{"frame":["claude: started","claude: one","claude: two","claude: three","claude: four","auto mode on (shift+tab to cycle)","$"],"clear":true},{"waitFile":"go3"},{"frame":["claude: started","claude: one","claude: two","claude: three","claude: four","claude: five","auto mode on (shift+tab to cycle)","$"],"clear":true},{"waitFile":"go4"},{"frame":["claude: started","claude: one","claude: two","claude: three","claude: four","claude: five","you: hello there","claude: six","auto mode on (shift+tab to cycle)","$"],"clear":true},{"waitFile":"go5"},{"frame":["claude: started","claude: one","claude: two","claude: three","claude: four","claude: five","you: hello there","claude: six","you: and again","claude: seven","auto mode on (shift+tab to cycle)","$"],"clear":true},{"idle":true}]}

start claude {temp}\order --claude {temp}\claude.cmd -- --model haiku --effort low
expect said Ready
file order\go1 x
do Conversation
expect text "claude: two","claude: three"

# Check: What Claude writes waits while a line is selected
# The caret at the top and a line selected: what Claude writes now waits (FR-3.11).
keys ctrl+home
keys shift+down
file order\go2 x
wait 2500
file order\go3 x
wait 2500
refuse text claude: four
# The selection cleared: both frames arrive at once, each line in its place, the caret still on the first line.
keys escape
expect said ! Selection cleared
expect text "claude: three","claude: four","claude: five"
expect text "caretLine":0

# Check: A message sent from the top of the conversation goes to its end
# A message sent with the caret left at the top of the conversation: its block stands at the end, after the last
# line, and the reply after it.
do Message field
type hello there
keys enter
expect said Sent
file order\go4 x
do Conversation
expect text "claude: five","# Input 1","hello there","","# Output 1 Reply from Claude","","claude: six"
refuse text "hello there","","# Output 1 Reply from Claude","","claude: one

# Check: A message from a quoted line goes to the end too
# The same from a quoted line: Enter on the first line copies it into the message, and the message goes to the end.
keys ctrl+home
keys enter
expect said ! Line 1 copied
type and again
keys enter
expect said Sent
file order\go5 x
do Conversation
expect text "claude: six","# Input 2","_ start of copied line from conversation history _","Line 1 of 12:","claude: started","_ start of comment on the copied line _","and again","","# Output 2 Reply from Claude",""
# The stand-in's own row for the message is no echo of the several lines sent, so it stays; the reply ends the text.
expect text "# Output 2 Reply from Claude","","you: and again","claude: seven"]
