# Workflow: Block Claude Code's feedback requests from the Options menu, and turn it off again
# axClaude: Claude Code's own settings (docs/axClaude/SPEC.md FR-14; nvda-test-plan section 21). Options, Claude Code
# settings writes one key of Claude Code's user settings file and keeps the rest of the file; off removes the key,
# which is Claude Code's own default. In a run the file is the run's own, home\.claude\settings.json, never the
# user's. Needs no Claude Code, no network and no model: a stand-in batch file draws one screen and waits, as in
# 25-edit-modes.
file home\AxKit\axClaude.json {"replySpeech":"none","interruptForQuestions":true,"workingMusic":false,"backgroundAlert":false,"checkForUpdates":false}
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"}]}
# A settings file with keys of the user's own, which must come through every change as they were.
file home\.claude\settings.json {\n  "theme": "dark",\n  "permissions": { "allow": ["Bash(git status)"] }\n}
mkdir cs
file claude.cmd @echo off\nif "%~1"=="--version" goto version\nchcp 65001 >nul\nfor /f %%a in ('echo prompt $E^| cmd') do set "E=%%a"\necho %E%[?25l%E%[2J%E%[Hclaude: started\necho auto mode on (shift+tab to cycle)\n<nul set /p "=$%E%[?25h"\n:idle\nping -n 2 127.0.0.1 >nul\ngoto idle\n:version\necho 0.0.0 (stand-in)

start claude {temp}\cs --claude {temp}\claude.cmd -- --model haiku
expect said Ready

# Check: Feedback requests are not blocked at first
# Off at first: Claude Code's own experience, the user's decision of 2026-11-15.
expect commands feedback requests, Options, Claude Code settings
refuse commands feedback requests, Options, Claude Code settings, checked
expect commands Choose the model, Actions
expect commands Choose the effort, Actions

# Goal: Block Claude Code's feedback requests
# On: the key written, "applies at once", the item checked, the user's keys kept, the old file kept beside it.
menu Block Claude Code's feedback requests
expect said ! Claude Code's feedback requests blocked, applies at once
expect commands feedback requests, Options, Claude Code settings, checked
run findstr /c:"feedbackSurveyRate\": 0" home\.claude\settings.json
run findstr /c:"Bash(git status)" home\.claude\settings.json
run findstr theme home\.claude\settings.json
run findstr theme home\.claude\settings.json.axkit-backup
run findstr feedbackSurveyRate home\.claude\settings.json.axkit-backup & if errorlevel 1 (exit 0) else (exit 1)

# Goal: Turn the block on feedback requests off again
# Off: the key removed, Claude Code's default back, the rest still there.
menu Block Claude Code's feedback requests
expect said ! Claude Code's feedback requests back to Claude Code's own default, applies at once
refuse commands feedback requests, Options, Claude Code settings, checked
run findstr feedbackSurveyRate home\.claude\settings.json & if errorlevel 1 (exit 0) else (exit 1)
run findstr /c:"Bash(git status)" home\.claude\settings.json

# Goal: Choose how Claude-drafted feedback comes
# Claude-drafted feedback, a choice of three: quiet written, then the default, which removes the key again.
menu Claude-drafted feedback: quiet
expect said ! Claude-drafted feedback: quiet, counted without a card, applies at once
expect commands Claude-drafted feedback: quiet, counted without a card, Options, Claude Code settings, checked
run findstr /c:"feedbackDrafts\": \"quiet\"" home\.claude\settings.json
menu Claude-drafted feedback: a card for each
expect said ! Claude-drafted feedback: a card for each, Claude Code
run findstr feedbackDrafts home\.claude\settings.json & if errorlevel 1 (exit 0) else (exit 1)

# Check: A settings file that is not valid JSON is never written
# A file that is not valid JSON is never written: the error says so and the file stays as it was.
file home\.claude\settings.json { "theme": "dark", }
allow error Claude Code's settings file was not changed
menu Block Claude Code's feedback requests
expect state "notice":"Error"
expect state not valid JSON at line 1
keys escape
run findstr feedbackSurveyRate home\.claude\settings.json & if errorlevel 1 (exit 0) else (exit 1)
