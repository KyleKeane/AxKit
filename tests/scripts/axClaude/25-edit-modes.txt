# Workflow: Read back a long message before sending it, and turn the music on with Ctrl+M
# axClaude: the bundle's mode key, Ctrl+Space (docs/axKit/SPEC.md AX-8.13; docs/axClaude/SPEC.md §6.3; nvda-test-plan
# section 6). In the message field it toggles navigation mode, where the letters move by the message's headings, links
# and lists and nothing changes the message, Enter included; in the conversation, which cannot be edited, it says so.
# The music key is Ctrl+M since 2026-11-15 (FR-7.9). Needs no Claude Code, no network and no model: a stand-in batch
# file draws one screen and waits, as in 23-key-test.
file home\AxKit\axClaude.json {"checkForUpdates":false}
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"}]}
mkdir em
file claude.cmd @echo off\nif "%~1"=="--version" goto version\nchcp 65001 >nul\nfor /f %%a in ('echo prompt $E^| cmd') do set "E=%%a"\necho %E%[?25l%E%[2J%E%[Hclaude: started\necho auto mode on (shift+tab to cycle)\n<nul set /p "=$%E%[?25h"\n:idle\nping -n 2 127.0.0.1 >nul\ngoto idle\n:version\necho 0.0.0 (stand-in)

start claude {temp}\em --claude {temp}\claude.cmd -- --model haiku
expect said Ready
do Message field
expect focus Message
# A message of a few lines with a heading, a link and a list (Shift+Enter makes a new line).
type # Plan
keys shift+enter shift+enter
type See https://example.com first.
keys shift+enter shift+enter
type - one
keys shift+enter
type - two
keys ctrl+home
refuse state navigation mode

# Goal: Move through a message by its headings, links and lists
# On: said, and the status bar says the field is in navigation mode.
keys ctrl+space
expect said ! Navigation mode
expect state field in navigation mode
# The letters move by the message's structure.
keys k
expect said ! https://example.com, link
keys i
expect said ! one
keys i
expect said ! two
keys shift+h
expect said ! Plan, heading level 1
# Check: Enter and letters that are no jump change nothing in navigation mode
# Enter does not send and a letter that is no jump types nothing: the edge sound, the message as it was.
keys enter
expect said ♪ Edge
keys z
expect said ♪ Edge
refuse said Sent
expect focus {"name":"Message to Claude","role":"Text","value":"# Plan"}

# Check: The conversation says it has no edit mode
# The conversation cannot be edited: the key says so and the mode of the field stays.
do Conversation
expect focus Conversation
keys ctrl+space
expect said ! Read only: there is no edit mode here
expect state field in navigation mode

# Goal: Go back to editing the message
# Back in the field, Ctrl+Space again: edit mode, and typing types.
do Message field
keys ctrl+space
expect said ! Edit mode
refuse state navigation mode
keys end
type !
expect focus {"name":"Message to Claude","role":"Text","value":"# Plan!"}

# Goal: Turn the background music on and off
# The music moved to Ctrl+M: from the field it is taken before the field (no line break), and from the conversation.
do Background music on or off
expect said ! Background music on, nothing is running
expect focus {"name":"Message to Claude","role":"Text","value":"# Plan!"}
do Conversation
do Background music on or off
expect said ! Background music off
# Check: The key test names the mode key and the music key
# The key test names both keys.
menu Test a key
expect said ! Key test.
keys ctrl+m
expect said ! Ctrl+M: Background music on or off
keys ctrl+space
expect said ! Ctrl+Space: Edit mode or navigation mode: says this text is read only
keys escape
expect said ! Key test ended
