# Workflow: Use axMedia's Context menu
# axMedia, plan step 1.9 (docs/axMedia/nvda-test-plan.md; SPEC.md AM-5.1, bundle AX-9.11): Context holds what acts on
# the file selected in the queue, Play, Remove and Information, and one item, "Nothing in the queue", while the queue
# is empty. The commands are run by name through the command list and found in the command palette, since Shift+F10
# and Alt+C open the menu in menu mode, which only the foreground window enters.
# Every sound is the runner's silence; nothing reaches the speakers.

# The run's own settings (AXKIT_HOME): no update check at start, which goes to the network.
file home\AxKit\axMedia.json {"checkForUpdates":false}

# An empty queue: Context's one item says that there is nothing to act on, and says it again when chosen.
start media
expect focus Queue is empty
expect commands "Nothing in the queue, Context"
refuse commands "Play, Context
menu Nothing in the queue, Context
expect said ! Nothing in the queue
close
wait 1500

# A queue of two: Context is the selected file's.
mkdir ch
silence ch\alpha.wav 30000
silence ch\beta.wav 30000
start media {temp}\ch
expect said ! Playing alpha.wav
expect focus Queue, 2 items
expect commands "Play, Context, Enter","Remove, Context, Delete","Information, Context, Ctrl+I"
refuse commands "Nothing in the queue, Context"

# Information on the playing file, with its position; on the other, its place only.
menu Information, Context
expect said ! alpha.wav, 1 of 2, playing, 0:0
keys down
expect focus beta.wav
menu Information, Context
expect said ! beta.wav, 2 of 2
refuse said playing,

# Play on the selected file: it plays and carries the mark.
menu Play, Context
expect said ! Playing beta.wav
expect list ["alpha.wav","beta.wav (playing)"]
expect state "title":"beta.wav - axMedia"

# Remove, found in the palette: its name, its menu and its key are said; Enter removes the selected file, which was
# playing, and the queue's last file gone, nothing plays.
keys ctrl+shift+p
expect state "notice":"Command palette"
type remove
expect said Remove, Context, Delete
keys enter
expect state "notice":null
expect said ! Removed beta.wav
expect list "name":"Queue, 1 item"
expect list "items":["alpha.wav"]

# Remove on the last file: the queue is empty, and Context says so again.
menu Remove, Context
expect said ! Removed alpha.wav
expect list ["Queue is empty"]
expect commands "Nothing in the queue, Context"
