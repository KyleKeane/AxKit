# Enter keeps the programs off and the Changes list comes; opened again, Trust this repository says "Trusted", and the
# opening after that shows no notice.
# The last check has the window refuse trusts written into its settings file: that refusal is logged as an error.

run git init -q -b main repo
file repo\a.txt hello
run git -C repo add .
run git -C repo -c user.name=Test -c user.email=none commit -q -m first
file repo\a.txt hello\nchanged
file repo\.git\hooks\post-checkout #!/bin/sh\nexit 0

# Goal: Open a repository with its own programs kept off
# 1.7 The notice names the hook; Enter keeps the programs off, and the Changes list comes.
start git {temp}\repo
expect state "notice":"This repository runs programs of its own"
expect state Its settings or hooks name programs git would run: hook post-checkout. axGIT runs git without them.
expect focus "name":"This repository runs programs of its own"
expect text Enter or Escape: Keep them off.
# The notice holds the answering keys for a second after it appears (KeyHold).
wait 1200
keys enter
expect state "notice":null
expect focus Changes, 1 file
expect focus Modified a.txt, unstaged
refuse said Trusted
close

# Goal: Trust a repository
# Opened again: the notice again, since nothing was trusted; Trust this repository.
start git {temp}\repo
expect state "notice":"This repository runs programs of its own"
wait 1200
keys tab
expect focus Keep them &off
keys tab
expect focus &Trust this repository
keys space
expect said Trusted
expect state "notice":null
expect focus Changes, 1 file
close

# Check: A trusted repository opens without the notice
# The next opening shows no notice: the Changes list at once.
start git {temp}\repo
expect focus Changes, 1 file
expect state "notice":null
refuse said This repository runs programs of its own
close

# Check: A trusted repository that gains a hook is asked about again
# AG-1.6: the trust was for the hook the repository had; with one more, its programs are set aside again and the
# notice says that what it would run has changed.
file repo\.git\hooks\pre-commit #!/bin/sh\nexit 0
start git {temp}\repo
expect state "notice":"This repository runs programs of its own"
expect state What this repository would run has changed since you trusted it, so it is not trusted now. Its settings or hooks name programs git would run: hook post-checkout, hook pre-commit.
wait 1200
keys enter
expect state "notice":null
close

# Check: A program named in the work tree's own settings file or in a repository inside is found too
# AG-1.6 (the review of 2026-11-27): git also reads the work tree's own settings file (extensions.worktreeConfig)
# and, running itself inside a submodule, that repository's settings; until then only the repository's own file was
# asked about. Here an ssh command in the work tree's file, and a repository inside, vendor, that names a pager:
# both are named, the second with the repository it is in. The names are never run: every git command of this
# window sets them to nothing, and nothing here reaches a server.
run git -C repo config extensions.worktreeConfig true
run git -C repo config --worktree core.sshCommand never-run
run git init -q -b main repo\vendor
file repo\vendor\b.txt inside
run git -C repo\vendor add .
run git -C repo\vendor -c user.name=Test -c user.email=none commit -q -m inside
run git -C repo\vendor config core.pager never-run
run git -C repo -c advice.addEmbeddedRepo=false add vendor
start git {temp}\repo
expect state "notice":"This repository runs programs of its own"
expect state name programs git would run: core.sshcommand, hook post-checkout, hook pre-commit, submodule vendor: core.pager. axGIT runs git without them.
wait 1200
keys enter
expect state "notice":null
expect focus Changes,
close

# Check: Repositories another program trusted in the settings file are not used
# AG-8.1a and the bundle's repair R-53: the window sealed the file as it saved the trust; a file written over it
# without the seal trusts nothing, and the window says so in its Settings notice. Opened without a repository, so
# that no notice of the repository's own comes in front of it.
wait 2000
