# Workflow: Open a repository with settings axGIT does not know, and start no fetch or push git cannot name a server for
# Needs: Git installed.
# axGIT: the reviews' second to fourth passes (docs/axGIT/nvda-test-plan.md sections 7 to 9; SPEC AG-1.6 to AG-1.6c,
# AG-5.4 to AG-5.4c). Only a setting axGIT knows to be harmless is passed over: any other is named in the notice, here
# one that is no program, and nothing runs in the repository until it is trusted (T1). The check is made again at
# every refresh and before every fetch, pull and push (T4), so a hook that appears or changes while the window is open
# is asked about then. A push without a branch, a fetch from a remote git does not know (an address in place of a
# name), a fetch from an address in no form axGIT reads (D4), a pull of a branch that follows none (T3) and a push to
# a folder (T2) are not started, and a notice says why. Nothing here reaches a server: the one address is of the
# reserved name "invalid", and no command is run with it.
file home\AxKit\axGIT.json {"checkForUpdates": false}
# The access layers' reminders agreed to already (AX-11.5): this script is about what comes after them.
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"},{"app":"axGIT","target":"Git","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Windows PowerShell","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"PowerShell 7","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Command Prompt","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Git Bash","version":"0","when":"2026-10-31T00:00:00"},{"app":"axPass","target":"KeePassXC","version":"0","when":"2026-10-31T00:00:00"}]}
run git init -q -b main repo
file repo\a.txt hello
run git -C repo add .
run git -C repo -c user.name=Test -c user.email=none commit -q -m first
# A clone made before anything odd is set: an everyday repository, for the last check.
run git clone -q {temp}\repo clone
# A setting that runs no program and is in no table of the harmless ones.
run git -C repo config http.sslVerify false

# Goal: Keep a repository with a setting axGIT does not know off
# 7.1 The notice has its own title and names the setting; Enter keeps it off, and nothing is opened.
start git {temp}\repo
expect state "notice":"This repository has settings axGIT does not know"
expect state It has settings axGIT does not know to be harmless: http.sslverify.
expect state Keep it off closes the repository.
expect text Enter or Escape: Keep it off.
# The notice holds the answering keys for a second after it appears (KeyHold).
wait 1200
keys enter
expect said Not opened
expect state "notice":null
close

# Goal: Trust a repository with a setting axGIT does not know
start git {temp}\repo
expect state "notice":"This repository has settings axGIT does not know"
wait 1200
keys tab
expect focus Keep it &off
keys tab
expect focus &Trust this repository
keys space
expect said Trusted
expect focus Changes, no changes

# Check: A hook that appears while the window is open is asked about at the next refresh, and Keep it off closes
# AG-1.6b: the check was one look as the repository opened; now the hook is found before the lists are read, and
# nothing runs until the user chooses.
file repo\.git\hooks\pre-commit #!/bin/sh\nexit 0
keys f5
expect state "notice":"This repository runs programs of its own"
expect state What axGIT finds in this repository differs from what you trusted, so it is not trusted now. axGIT runs nothing more in this repository until you choose. Its settings or hooks name programs git would run: hook pre-commit. It has settings axGIT does not know to be harmless: http.sslverify.
wait 1200
keys enter
expect said Repository closed
expect state "notice":null
expect state "title":"axGIT"
close

# Check: A hook changed outside the window is asked about before a fetch
# AG-5.4 (T4): the check comes before every fetch, pull and push, so the fetch does not start, and the notice asks first.
start git {temp}\repo
expect state "notice":"This repository runs programs of its own"
wait 1200
keys tab
keys tab
keys space
expect said Trusted
expect focus Changes, no changes
file repo\.git\hooks\pre-commit #!/bin/sh\necho changed\nexit 0
keys ctrl+r
expect state "notice":"This repository runs programs of its own"
expect state What axGIT finds in this repository differs from what you trusted
wait 1200
keys tab
keys tab
keys space
expect said Trusted
expect state "notice":null

# Check: A push while HEAD is on no branch is not started
# 7.4 AG-5.4a: git, run without a remote, pushed where its settings said; axGIT pushes a branch, to the remote it asked about.
run git -C repo checkout -q --detach
keys f5
# The status bar shows the refresh is in: an older "Refreshed" would be matched by the words alone.
expect state detached
keys ctrl+p
expect state "notice":"Push not started"
expect state HEAD is on no branch (detached), and axGIT pushes a branch. Switch to a branch first. Nothing was sent and nothing was fetched.
wait 1200
keys enter
expect state "notice":null
run git -C repo checkout -q main

# Check: A fetch from a remote git does not know is not started
# 7.3 AG-5.4a: an address in place of a remote's name. Nobody could be asked, and the fetch ran all the same.
run git -C repo config branch.main.remote https://example.invalid/x.git
keys f5
expect state main · no upstream
keys ctrl+r
expect state "notice":"Fetch not started"
expect state Git could not say where the remote https://example.invalid/x.git leads.
expect state Nothing was sent and nothing was fetched.
wait 1200
keys enter
expect state "notice":null

# Check: An address in no form axGIT reads starts nothing
# AG-5.4a (the review's third pass, D4): c:nul looked like a folder, and git ran ssh to the host c with it.
run git -C repo config branch.main.remote origin
run git -C repo remote add origin c:nul
keys f5
expect state main · no upstream
keys ctrl+r
expect state "notice":"Fetch not started"
expect state The remote origin has an address axGIT does not read: c:nul.
wait 1200
keys enter
expect state "notice":null

# Check: A push to a folder is not started
# AG-5.4c (the fourth pass, T2): git's transport for a folder runs the other repository's receive hooks, and which
# repository it enters is git's to decide; axGIT pushes only to a server, and nothing arrives in the folder.
run git init -q --bare mirror.git
file mirror.git\hooks\pre-receive #!/bin/sh\nexit 0
run git -C repo remote set-url origin ../mirror.git
keys ctrl+p
expect state "notice":"Push not started"
expect state This remote is a folder; push to it in a terminal. Nothing was sent.
wait 1200
keys enter
expect state "notice":null
# Nothing arrived there.
run git -C mirror.git rev-parse --verify --quiet refs/heads/main && exit 1 || exit 0

# Check: A pull of a branch that follows none is not started
# AG-5.4 (T3): a pull names the branch it brings; this one follows no branch of the remote, so nothing is named.
keys ctrl+l
expect state "notice":"Pull not started"
expect state This branch follows no branch of a remote, so there is nothing named to pull. Nothing was fetched.
wait 1200
keys enter
expect state "notice":null
close

# Check: An everyday repository opens without the notice
# AG-1.6c: what git clone wrote (the remote, the branch it follows) is known to be harmless.
start git {temp}\clone
expect focus Changes, no changes
expect state "notice":null
refuse said This repository
close
