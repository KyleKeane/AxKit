# Workflow: Let an AI tool read and add tasks
# axTask's typed data tools (docs/axKit/mcp.md MC-8, docs/axTask/SPEC.md TK-8): the lists and the items read through
# the open axTask window, an item added there by words with a Due date, said in the window and seen in its list, taken
# back with Ctrl+Z; an add into another list and into a project's list, the selection left where it was; a list
# elsewhere refused; the add refused while Let AI tools run commands is off. The runner is the MCP client.
# A list made as a GitHub list (TK-7a.2): the "github" member of its list.json means nothing now; it is an ordinary list.
file tasks\Issues\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[],"actions":[],"github":{"repository":"owner/repo"}}
file tasks\Errands\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[],"actions":[]}
file projects\alpha\Home\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[{"title":"Done","type":"check","value":false}],"actions":[]}
file "projects\alpha\Home\Fix the tap.json" {"schema":"list.json","Title":"Fix the tap","Done":false}
file home\AxKit\axTask.json {"checkForUpdates": false}
# The settings file names the project folder by its full path, which only the run knows.
run powershell -NoProfile -Command "[IO.File]::WriteAllText('home\AxKit\axTask.json', (ConvertTo-Json -InputObject @{checkForUpdates=$false; projectFolders=@((Resolve-Path 'projects').Path)}))"
file tasks\Personal\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[{"title":"Done","type":"check","value":false},{"title":"Due","type":"date","value":""},{"title":"Priority","type":"choice","value":"Normal","options":["High","Normal","Low"]}],"actions":[]}
file "tasks\Personal\Buy milk.json" {"schema":"list.json","Title":"Buy milk","Done":false,"Due":""}
file "tasks\Personal\Old chore.json" {"schema":"list.json","Title":"Old chore","Done":true}
start task {temp}\tasks
expect state "title":"Lists, axTask
type pe
expect focus Personal, list, 2 items
keys right
expect focus Buy milk

# The access on and the runner paired, as in 01-read-and-pair.
menu Let AI tools read and open axKit
expect state "notice":"Turn on AI tools?"
keys enter
expect said AI tools on
try said Not in a test run
mcp start
mcp call axkit_task_lists
expect state "notice":"axAutomate asks to use axKit"
wait 1500
keys tab
expect focus I understand
keys space
expect said ! checked
keys tab
expect focus &Allow
keys enter
expect said axAutomate may read axKit

# Check: The call that waited gives the lists with their counts
# TK-8.1: the lists with their counts, read without a key.
expect mcp "notDone": 1

# Goal: Let an AI tool read the task lists
# TK-8.1, MC-D14: a project's list as "project / list", with the project as its place, once the projects are found.
wait 1000
mcp call axkit_task_lists
expect mcp "name": "alpha / Home"
mcp call axkit_task_lists
expect mcp "place": "alpha"

# Goal: Let an AI tool read a list's items
# TK-8.1: one list's items with their fields; the done ones only when asked for.
mcp call axkit_task_items {"list":"personal"}
expect mcp "doneLeftOut": 1
mcp call axkit_task_items {"list":"Personal","done":true}
expect mcp "title": "Old chore"

# Check: A list outside the window's folders is refused
# MC-D12: a list named by a path outside the window's folders is refused.
mcp call axkit_task_items {"list":"D:\\Elsewhere\\Personal"}
expect mcp is not one of the axTask window's lists

# Check: Adding is refused while AI tools may not run commands
# MC-8.3: adding is a change, refused while Let AI tools run commands is off, and the window says nothing.
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Personal","title":"Call the dentist","fields":{"Due":"2026-11-18"}}
expect mcp error: The user has not allowed AI tools to run commands in axKit
refuse said added Call the dentist

# Goal: Let AI tools run commands
menu Let AI tools run commands
expect said AI tools may run commands: each is said as it runs

# Check: A value the field does not take is refused
# TK-8.3: a value the field does not take is refused with its reason, and nothing is written.
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Personal","title":"Call the dentist","fields":{"Priority":"Urgent"}}
expect mcp error: Priority: Not one of the options: High, Normal, Low

# Goal: Let an AI tool add a task
# TK-8.3, MC-8.4: added from the list's template with the Due given, said in the window, selected in the list shown.
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Personal","title":"Call the dentist","fields":{"Due":"2026-11-18","Priority":"High"}}
expect mcp "added": "Call the dentist"
expect said axAutomate added Call the dentist to Personal
expect focus Call the dentist
run findstr /c:"2026-11-18" "tasks\Personal\Call the dentist.json"
run findstr /c:"High" "tasks\Personal\Call the dentist.json"

# Goal: Undo a task an AI tool added
# One step of the window's undo: Ctrl+Z takes it back.
keys ctrl+z
expect said Undone: axAutomate added Call the dentist to Personal
refuse focus Call the dentist

# Goal: Let an AI tool add a task to another list
# TK-8.4, MC-D15: an add into a list not shown is said, and the selection stays where the user left it.
keys home
expect focus Buy milk
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Errands","title":"Post the parcel"}
expect mcp "added": "Post the parcel"
expect said axAutomate added Post the parcel to Errands
expect focus Buy milk
run findstr /c:"Post the parcel" "tasks\Errands\Post the parcel.json"

# Goal: Let an AI tool add a task to a project's list
# TK-8.2: a project's list by its own name alone, since only one list has it.
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"home","title":"Buy a washer"}
expect said axAutomate added Buy a washer to alpha / Home
expect focus Buy milk
run findstr /c:"Buy a washer" "projects\alpha\Home\Buy a washer.json"

# Check: A list made as a GitHub list takes an add like any other
# TK-8.3, TK-7a.2 (2026-11-14): axTask has no GitHub lists any more, so an add into a list made as one is taken as
# into any list (from 2026-11-12 it was refused).
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Issues","title":"A new issue"}
expect said axAutomate added A new issue to Issues
run findstr /c:"A new issue" "tasks\Issues\A new issue.json"

# Goal: Stop AI tools from running commands
# Off again from the menu: the next add is refused at once.
menu Let AI tools run commands
expect said AI tools may no longer run commands
# Under NVDA the keys are real input: the window in front changes only after 3 s without one (MC-7.8).
wait 3500
mcp call axkit_task_add {"list":"Personal","title":"Call the dentist"}
expect mcp error: The user has not allowed AI tools to run commands

# Check: The adds are in the log
# MC-6.6: the calls are in the log by their tool.
run findstr /c:"axkit_task_add" home\AxKit\mcp-audit.log

# Check: With no axTask window open the tools say how to open one
# MC-D10: the tools work through an open axTask window; with none open, nothing happens and the answer says how to open one.
keys alt+f4
expect closed
mcp call axkit_task_lists
expect mcp error: No axTask window is open
