# Workflow: Change an entry's fields, keep a backup and delete for good
# Needs: KeePassXC installed in C:\Program Files\KeePassXC (the script makes its test database with keepassxc-cli.exe there).
# axPass, docs/axPass/user-guide.md beyond 01 to 03: "Changing entries" (the title, the URL and the notes changed in
# a notice, a password typed twice, which Ctrl+Z does not bring back, Session, Back up now and Show backups in axFile,
# a new group, a group renamed in KeePassXC's own window, a group moved to the Recycle Bin, an entry deleted for good
# there), "The window" (Enter opens a group, Backspace goes back up, Right and Left change nothing in the fields) and
# "Using a password" (Ctrl+U on a URL that is no web address opens nothing); SPEC PW-4.5, PW-6, PW-7. The database is made
# here by KeePassXC's own tool, the master password and every value a test value. The copies (Ctrl+B, Ctrl+C) stay
# with the NVDA pass, since they would change the clipboard of whoever runs this, and Ctrl+U on a web address is not
# pressed: it would start the browser.
# Not here: a field emptied (Enter on it, its text deleted, Enter), and Ctrl+Z after notes were written into an empty
# Notes field. Both end today in the words of KeePassXC's tool, "Not changing any field for entry ...", and the field
# keeps its text; that is the report's to name, not a step to write as passing. So the notes are written first here,
# and the change Ctrl+Z takes back is the URL's, from one address to another.
# The idle lock off: it counts real input only, which the runner's keys are not.
file home\AxKit\axPass.json {"checkForUpdates": false, "lockMinutes": 0}
file home\AxKit\axFile.json {"checkForUpdates": false}
# The access layers' reminders agreed to already (AX-11.5): this script is about what comes after them.
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"},{"app":"axGIT","target":"Git","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Windows PowerShell","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"PowerShell 7","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Command Prompt","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Git Bash","version":"0","when":"2026-10-31T00:00:00"},{"app":"axPass","target":"KeePassXC","version":"0","when":"2026-10-31T00:00:00"}]}
file twice.txt test-master-1\ntest-master-1\n
file once.txt test-master-1\n
run "C:\Program Files\KeePassXC\keepassxc-cli.exe" db-create -q -p test.kdbx < twice.txt
run "C:\Program Files\KeePassXC\keepassxc-cli.exe" mkdir -q test.kdbx Email < once.txt
run "C:\Program Files\KeePassXC\keepassxc-cli.exe" mkdir -q test.kdbx Work < once.txt
run "C:\Program Files\KeePassXC\keepassxc-cli.exe" add -q -u sam --url https://mail.example.com test.kdbx Email/Gmail < once.txt
# An entry whose URL is a folder's path, no web address.
run "C:\Program Files\KeePassXC\keepassxc-cli.exe" add -q -u kim --url C:\made-up\share test.kdbx Email/Share < once.txt

start pass {temp}\test.kdbx
expect state Unlock test
type test-master-1
keys enter
expect said ! test, 2 items, Email, group, 2 items

# Check: Before any change there is no backup to show
# Session, Show backups in axFile right after unlocking: there is no folder of backups yet, and that is said.
do Show backups in axFile
expect said ! No backups yet: the first change after unlocking makes one, or Session, Back up now

# Goal: Make a backup now
# Session, Back up now: the copy's name, with the date and time, and its folder said; the file is there.
do Back up now
expect said ! Backed up: test
expect said , in test backups
run cmd /c dir /b "test backups\test *.kdbx"

# Goal: Show the backups in axFile
# Session, Show backups in axFile: an axFile window on the folder "test backups" (a window of this run).
do Show backups in axFile
adopt
expect state "title":"test backups - axFile"
close
use 1

# Check: A group is not renamed here
# F2 on a group: a group is renamed in KeePassXC's own window, and that is said; no rename box opens.
keys f2
expect said ! A group is renamed in KeePassXC's own window
expect focus Email, group, 2 items

# Goal: Open a group with Enter and go back up with Backspace
keys enter
expect said ! Email, 2 items, Gmail
keys backspace
expect said ! test, 2 items, Email, group, 2 items
keys enter
expect said ! Email, 2 items, Gmail

# Goal: Write an entry's notes over two lines
# Enter on Notes: a field of several lines, where Enter makes a new line (said, AX-7.8). Tab leaves the field (the
# focus is then on the notice's own text), and Enter there is Save, the notice's default: the notes are kept.
keys enter
expect focus Title, text, Gmail
keys end
expect focus Notes, long text, empty
keys enter
expect state "notice":"Change Notes"
expect focus {"name":"Notes","role":"Text","value":""}
type First line of the notes
keys enter
expect said new line
type Second line
keys tab
keys enter
expect said ! Notes: First line of the notes, and 1 more line
expect focus Notes, long text, First line of the notes

# Goal: Change an entry's URL
# Enter on the URL field: the notice "Change URL" with the address in its field; a new one typed over it, Enter.
keys up
expect focus URL, text, https://mail.example.com
keys enter
expect state "notice":"Change URL"
expect focus {"name":"URL","role":"Text","value":"https://mail.example.com"}
type https://example.org/login
keys enter
expect said ! URL: https://example.org/login
expect focus URL, text, https://example.org/login

# Check: Right and Left change nothing in the fields and play the edge sound
keys right
expect said ♪ Edge
keys left
expect said ♪ Edge
expect focus URL, text, https://example.org/login

# Goal: Set a password typed twice
# Enter on Password: a new one typed, then typed again; the same twice sets it. The field hides what is typed.
keys up
expect focus Password, password, hidden
keys enter
expect focus Type a new password for Gmail
type made-up-pass-1
keys enter
expect focus Re-enter the password for Gmail
type made-up-pass-1
keys enter
expect said ! Password set for Gmail
expect focus Password, password, hidden

# Check: Ctrl+Z does not bring an old password back
# Ctrl+Z right after the new password takes back the change before it, the URL, and never the password.
keys ctrl+z
expect said ! Undone: URL: https://example.org/login
keys down
expect focus URL, text, https://mail.example.com

# Goal: Change an entry's title in its fields
# Enter on Title: the notice "Change Title" with the title in its field; the new one renames the entry.
keys home
expect focus Title, text, Gmail
keys enter
expect state "notice":"Change Title"
expect focus {"name":"Title","role":"Text","value":"Gmail"}
type Gmail work
keys enter
expect said ! Renamed Gmail work
expect focus Title, text, Gmail work

# Check: A URL that is no web address is not opened
# Ctrl+U on the entry whose URL is a folder's path: nothing is opened, and that is said.
keys escape
expect focus Gmail work, sam
wait 1100
type sh
expect focus Share, kim
keys ctrl+u
expect said ! Not opened: Share's URL is not a web address

# Goal: Move an entry to the Recycle Bin
# Delete on the entry: it goes to the Recycle Bin group, where the last sections find it.
keys delete
expect said ! Moved Share to the Recycle Bin

# Goal: Make a new group
# Ctrl+Shift+N at the top of the database: the notice "New group" with the field Name.
keys backspace
expect said ! test, 3 items, Email, group, 1 item
keys ctrl+shift+n
expect state "notice":"New group"
expect focus "name":"Name"
type Archive
keys enter
expect said ! Made Archive
expect list Archive, group

# Goal: Move a group to the Recycle Bin
# Delete on a group asks first, Cancel the default (01 checks that); Alt+M, the Move button, moves it.
wait 1100
type w
expect focus Work, group
keys delete
expect state "notice":"Move the group"
expect state Move the group Work and its 0 items to the Recycle Bin? Ctrl+Z cannot bring a group back
# A notice holds its answering keys for a moment after it appears (KeyHold).
wait 1200
keys alt+m
expect said ! Moved Work to the Recycle Bin
refuse list Work, group

# Check: The Recycle Bin itself is not deleted
wait 1100
type r
expect focus Recycle Bin, group
keys delete
expect said ! The Recycle Bin stays; KeePassXC empties it
expect state "notice":null

# Check: Deleting for good asks first and Cancel is the default
# In the Recycle Bin, Delete on an entry asks, naming it; Enter is Cancel, and the entry stays.
keys right
expect said ! Recycle Bin, 2 items
wait 1100
type sh
expect focus Share
keys delete
expect state "notice":"Delete for good"
expect state Delete Share for good? It is in the Recycle Bin, and nothing brings it back.
wait 1200
keys enter
expect state "notice":null
expect focus Share
expect list Share

# Goal: Delete an entry for good from the Recycle Bin
# Delete again, Alt+D, the Delete button: the entry is gone from the Recycle Bin.
keys delete
expect state "notice":"Delete for good"
wait 1200
keys alt+d
expect said ! Deleted Share
refuse list Share
