# Workflow: Fill in a number and a choice, and start a server again
# axMCP, docs/axMCP/user-guide.md beyond 01 and 02: "The servers" (Enter starts a server; Backspace goes back up and
# the server keeps running; Ctrl+Shift+S stops the server selected at the top; the agreement is for exactly what
# would run, so a changed argument asks again), "Tools" (a number; a choice stepped through with Right and Left and
# chosen from its list; h for the headings of a result and Ctrl+F to find in it; the values typed stay, so a second
# call is one key) and "If something goes wrong" (a server that stops by itself is said with its exit code, and Enter
# on its level starts it again; a server that does not start says why in a notice); also the names of a server's
# environment variables read in its details and in the agreement, never their values, and that closing the window
# stops a running server; SPEC MP-2, MP-3, MP-10.
# The server is a fake in Windows PowerShell, written here, as in 01-start: it answers the handshake and two tools,
# order (a number and a choice, its answer a text with two headings) and quit (the fake ends itself with exit code
# 3), and writes the file ended.txt when its input closes, which is how a client stops it. It is the server of the
# run folder's own .mcp.json (Claude Code's project scope), so the run neither needs nor changes Claude Code's files,
# and its definition can change under the window as a project's does; its row is found by name, since this computer
# may have servers of its own. Its definition sets one environment variable, a made-up name with a made-up value.
# The second server of that file names a program no computer has, and so never starts.
file home\AxKit\axMCP.json {"checkForUpdates": false}
file .mcp.json {"mcpServers":{"zzfake":{"command":"powershell","args":["-NoProfile","-ExecutionPolicy","Bypass","-File","fake.ps1"],"env":{"ZZ_MADE_UP_TOKEN":"made-up-secret-value"}},"zzmissing":{"command":"axkit-no-such-program-zz.exe","args":["one"]}}}
file fake.ps1 $in = [Console]::In\n$out = [Console]::Out\n$nl = [char]92 + 'n'\nwhile ($true) {\n  $line = $in.ReadLine()\n  if ($null -eq $line) { break }\n  if ($line.Trim().Length -eq 0) { continue }\n  $m = $line | ConvertFrom-Json\n  if ($null -eq $m.id) { continue }\n  $r = $null\n  switch ($m.method) {\n    'initialize' { $r = '{"protocolVersion":"2025-06-18","capabilities":{"tools":{},"resources":{},"prompts":{}},"serverInfo":{"name":"fake","version":"1"}}' }\n    'tools/list' { $r = '{"tools":[{"name":"order","description":"Takes an order. Nothing is sent anywhere.","inputSchema":{"type":"object","properties":{"count":{"type":"integer","description":"How many"},"size":{"type":"string","enum":["small","medium","large"]}},"required":["count"]}},{"name":"quit","description":"Ends the server.","inputSchema":{"type":"object","properties":{}}}]}' }\n    'tools/call' {\n      if ($m.params.name -eq 'quit') { exit 3 }\n      $r = '{"content":[{"type":"text","text":"# Order' + $nl + $nl + 'You asked for ' + $m.params.arguments.count + ' of size ' + $m.params.arguments.size + '.' + $nl + $nl + '## Thanks' + $nl + $nl + 'The order is made up."}]}'\n    }\n    'resources/list' { $r = '{"resources":[]}' }\n    'resources/templates/list' { $r = '{"resourceTemplates":[]}' }\n    'prompts/list' { $r = '{"prompts":[]}' }\n  }\n  if ($null -eq $r) { $out.WriteLine('{"jsonrpc":"2.0","id":' + $m.id + ',"error":{"code":-32601,"message":"Method not found"}}') } else { $out.WriteLine('{"jsonrpc":"2.0","id":' + $m.id + ',"result":' + $r + '}') }\n}\nSet-Content -Path ended.txt -Value 'the input closed'\n

start mcpclient {temp}
expect state Servers - axMCP, access layer for MCP servers
expect focus Servers

# Goal: Read the names of a server's environment variables
# Tab on the server: its details name the variable its definition sets, and never its value.
type zzfake
expect focus zzfake, stdio, stopped
keys tab
expect focus Source: Claude Code, project scope
expect list "Environment: 1 variable, ZZ_MADE_UP_TOKEN"
refuse list made-up-secret-value
keys escape

# Goal: Start a server with Enter
# Enter on a server starts it, as Right does: the agreement first (MP-2.1a), which names the environment variable
# and never its value, then the server's announcement.
keys enter
expect said Before axMCP starts zzfake
expect state Its definition sets the environment variables ZZ_MADE_UP_TOKEN; their values are not read out, since they may be tokens.
refuse state made-up-secret-value
keys tab
expect said I understand that axMCP
keys space
expect said ! checked
keys enter
expect said ! zzfake, running, Tools 2, Resources 0, Prompts 0
expect focus Tools, 2

# Check: Backspace goes back up and the server keeps running
keys backspace

# Goal: Stop the server selected at the top level

# Check: A server agreed to once starts again without the question
# The agreement is kept: Enter starts the server at once. Had the question come, nothing would start.
