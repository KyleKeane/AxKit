# Workflow: Start an MCP server and use its tools
# axMCP: the servers, a server started, its tool called, its resource read, stopped (docs/axMCP/nvda-test-plan.md
# sections 1 to 4 and 6), run by tools/AxAutomate on a desktop of its own (docs/axKit/automation.md). The server is a
# fake in Windows PowerShell, written here: it answers the handshake, one tool (echo) and one resource. The agreement
# for it is given in the run, since it is for the whole command line and the folder (MP-2.1a); the server is one of Your servers, so
# the run never touches Claude Code's files, and its row is found by name since this computer may have servers of its own.
file home\AxKit\axMCP.json {"servers": [{"name": "fake", "commandLine": "powershell -NoProfile -ExecutionPolicy Bypass -File fake.ps1"}]}
file fake.ps1 [Console]::Error.WriteLine('axkitfake warns of nothing')\n$in = [Console]::In\n$out = [Console]::Out\nwhile ($true) {\n  $line = $in.ReadLine()\n  if ($null -eq $line) { break }\n  if ($line.Trim().Length -eq 0) { continue }\n  $m = $line | ConvertFrom-Json\n  if ($null -eq $m.id) { continue }\n  $r = $null\n  switch ($m.method) {\n    'initialize' { $r = '{"protocolVersion":"2025-06-18","capabilities":{"tools":{},"resources":{},"prompts":{}},"serverInfo":{"name":"fake","version":"1"}}' }\n    'tools/list' { $r = '{"tools":[{"name":"echo","description":"Says the text back. Nothing else.","inputSchema":{"type":"object","properties":{"text":{"type":"string","description":"What to say"},"loud":{"type":"boolean"}},"required":["text"]}}]}' }\n    'tools/call' { $t = [string]$m.params.arguments.text; if ($m.params.arguments.loud) { $t = $t.ToUpper() }; $r = '{"content":[{"type":"text","text":"' + $t + '"}]}' }\n    'resources/list' { $r = '{"resources":[{"name":"readme","uri":"fake://readme","mimeType":"text/plain","description":"A greeting."}]}' }\n    'resources/templates/list' { $r = '{"resourceTemplates":[]}' }\n    'resources/read' { $r = '{"contents":[{"uri":"fake://readme","mimeType":"text/plain","text":"Hello from the fake server."}]}' }\n    'prompts/list' { $r = '{"prompts":[]}' }\n  }\n  if ($null -eq $r) { $out.WriteLine('{"jsonrpc":"2.0","id":' + $m.id + ',"error":{"code":-32601,"message":"Method not found"}}') } else { $out.WriteLine('{"jsonrpc":"2.0","id":' + $m.id + ',"result":' + $r + '}') }\n}\n
