# Workflow: Find and paste something you copied earlier
# axClip: the search field, the match keys and the commands that stay inside the window (docs/axClip/nvda-test-plan.md;
# SPEC CP-2, CP-4, CP-5.2). Every key here is a command of the window's one list (docs/axKit/app-kit.md AK-2), the match
# keys HotkeyForm's, so this script is also the check that the list does what the key switch did before it. The hotkey
# is one nobody uses, so that the run never takes the user's; Start with Windows stays on, which a development build
# never writes (SignInStart), since off would remove the user's own sign-in value.
#
# The test desktop shares the clipboard with the user's session: the script copies three texts of its own, which stay
# on the clipboard after the run, and every step filters on them, so that nothing the user copies is ever said here.
file home\AxKit\axClip.json {"checkForUpdates": false, "hotkey": "Ctrl+Alt+Shift+F10", "sayFirstMatch": false}
file copy.ps1 # Copies its arguments, one line each, and reads the clipboard back: clip.exe gives up silently when another program holds the clipboard open (the NVDA pass, 2026-11-15).\n$v = $args -join "`r`n"\nfor ($i = 0; $i -lt 20; $i++) { try { Set-Clipboard -Value $v; if ((Get-Clipboard -Raw) -eq $v) { exit 0 } } catch { }; Start-Sleep -Milliseconds 100 }\nexit 1
# A copy goes through copy.ps1, which reads the clipboard back and tries again: clip.exe gave up without a word while
# the user's own axClip and NVDA read the same clipboard, and a copy was missing from the history (2026-11-15).

start clip
# Check: The window's command list holds every key of its own
# The window's command list: every key of its own in the F1 text, none shadowed (docs/axKit/app-kit.md AK-2.4).
check commands
expect focus Search the clipboard

run powershell -NoProfile -ExecutionPolicy Bypass -File copy.ps1 "axkitclip alpha"
wait 500
run powershell -NoProfile -ExecutionPolicy Bypass -File copy.ps1 "axkitclip beta"
wait 500
run powershell -NoProfile -ExecutionPolicy Bypass -File copy.ps1 "axkitclip gamma"
wait 500

# Goal: Move through the copied items
# CP-2.4: Down and Up move the selection, newest first, with the caret kept in the field; at an end the edge sound,
# and the item said.
type axkitclip
keys down
expect said axkitclip beta, 2 of 3
keys up
expect said axkitclip gamma, 1 of 3
keys up
expect said ♪ Edge
expect said axkitclip gamma, 1 of 3

# Goal: Jump to the newest or oldest item, or by a page
# Ctrl+End the oldest (Down there is the edge), Ctrl+Home the newest; Page Down and Page Up by a page.
do Last match
expect said axkitclip alpha, 3 of 3
keys down
expect said ♪ Edge
do First match
expect said axkitclip gamma, 1 of 3
keys pagedown
expect said axkitclip alpha, 3 of 3
keys pageup
expect said axkitclip gamma, 1 of 3

# Goal: Read an item's whole text
# CP-4.3: Ctrl+Q the whole text in a notice; Escape closes it.
do Read in full
expect state "notice":"Clipboard text, 1 line"
keys escape
expect state "notice":null

# Goal: Pause the recording and start it again
# Ctrl+P pauses the recording and starts it again.
do Pause recording
expect said ! Recording paused
expect state Paused
do Pause recording
expect said ! Recording
expect state Recording

# Goal: Remove an item
# Delete removes the selected item.
keys delete
expect said ! Removed
do Last match
expect said , 2 of 2

# Goal: Clear the clipboard history
# Ctrl+Shift+Delete asks first; Clear empties the history.
do Clear history
expect state "notice":"Clear the clipboard history?"
keys enter
expect said ! Clipboard history cleared
keys down
expect said ♪ Edge

# Enter (Paste into the program in front) and Shift+Enter (Copy without pasting) put an item on the clipboard and
# act outside the window: not run here, they stay for the NVDA plan.

# Check: The shortcuts end with the Notices section and give the field back
# F1: the shortcuts end with the Notices section every window of axKit has (2026-11-09); Escape closes the text and the
# focus is back in the field.
do Keyboard shortcuts
expect state "notice":"Keyboard shortcuts"
expect text Notices
keys escape
expect state "notice":null
expect focus Search the clipboard

# Goal: Hide the window and show it again
# Escape hides the window whatever the field holds; a second start shows it again with the field empty.
keys escape
launch clip
expect focus "name":"Search the clipboard","role":"Text","value":""

# Goal: Quit axClip
# Quit axClip, the one close that ends the process.
menu Quit axClip
expect closed
