# Workflow: Group lists into sections
# axTask: the top level's sections (docs/axTask/SPEC.md TK-3a, TK-D4, TK-D5): a list moved to a section of the user's
# own with Ctrl+M, a new section, the Section field, a new list joining the selected section, the Archived lists
# section, Ctrl+Up and Ctrl+Down across them, a folder inside a list not shown.
file home\AxKit\axTask.json {"checkForUpdates": false}
file tasks\Home\Tap.json {"format": "axkit-item 1", "fields": [{"title": "Title", "type": "text", "value": "Tap"}, {"title": "Done", "type": "check", "value": false}]}
file tasks\Shopping\Milk.json {"format": "axkit-item 1", "fields": [{"title": "Title", "type": "text", "value": "Milk"}]}
file tasks\Work\Report.json {"format": "axkit-item 1", "fields": [{"title": "Title", "type": "text", "value": "Report"}, {"title": "Done", "type": "check", "value": false}]}
file tasks\Work\Nested\Hidden.json {"format": "axkit-item 1", "fields": [{"title": "Title", "type": "text", "value": "Hidden"}]}

start task {temp}\tasks
expect focus Your lists, heading level 1, 3 lists

# Goal: Move a list into a new section
# TK-3a.3 Ctrl+M on Work: one form (bundle AX-8.10), the sections as a list, then the new section's name. A new
# section without a name is refused with the focus in the name's field; then Jobs.
type w
expect focus Work, list, 1 item
do Move to section
expect state "notice":"Move Work"
expect focus "name":"Choose the section"
expect focus Your lists
keys end
expect focus A new section
keys enter
expect said ♪ Edge
expect said ! Type the section's name
expect state "notice":"Move Work"
expect focus "name":"Type the new section's name, when you chose A new section"
type Jobs
keys enter
expect said ! Moved Work to Jobs
expect focus Work, list, 1 item
run findstr /l /c:"Jobs" tasks\Work\list.json
keys up
expect focus Jobs, heading level 2, 1 list

# Goal: Move among the sections by their headings
# TK-3a.2 Ctrl+Up back to Your lists; Ctrl+Down to Jobs, a heading of level 2, then Smart lists; Ctrl+1 to Date lists,
# the last of level 1: past it the edge sound; Ctrl+Shift+2 back to Jobs (TK-D14).
do Previous section
expect focus Your lists, heading level 1, 2 lists
do Next section
expect focus Jobs, heading level 2
do Next section
expect focus Smart lists, heading level 1, 1 smart list
do Next heading level 1
expect focus Date lists, heading level 1, 4 date lists
do Next heading level 1
expect said ♪ Edge
do Previous heading level 2
expect focus Jobs, heading level 2

# Goal: Make a new list in a section
# TK-3a.3 A new list joins the section selected.
do New list
type Clients
keys enter
expect said ! Made Clients
expect focus Clients, list, 0 items
keys up
expect focus Jobs, heading level 2, 2 lists

# Goal: Put a list in a section from its details
# The Section field: Home joins Jobs from its details, then back to Your lists with Ctrl+M.
do Previous section
keys down
expect focus Home, list, 1 item
keys tab
expect focus Section, text, empty
keys enter
type Jobs
keys enter
expect said ! Section: Jobs
keys escape
expect focus Home, list, 1 item
# Goal: Move a list back to Your lists
do Previous section
expect focus Jobs, heading level 2, 3 lists
type h
expect focus Home, list, 1 item
do Move to section
expect focus Jobs
keys home
keys enter
expect said ! Moved Home to Your lists

# Check: A folder inside a list is counted, not shown
# TK-D4: a folder inside a list is counted, not shown.
type w
expect focus Work, list, 1 item
keys right
expect said ! Work, 1 item, 1 file not shown, Report

# Goal: Archive a list and undo it
# IA-D4: an archived list goes to Archived lists, its own section.
keys left
keys delete
expect said ! Archived Work
keys end
expect focus Work, list, 1 item
keys up
expect focus Archived lists, heading level 1, 1 list
do Undo
expect said ! Undone: Archived Work
