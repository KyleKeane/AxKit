# Workflow: Keep the day each item was made
# axTask, docs/axTask/user-guide.md "The fields of a list" and SPEC.md TK-4.1a: a list whose configuration names a
# field Created, of type date, gets today's date there for every item added; the day itself changes with the
# calendar, so the file is asked for a date of this century.
file tasks\Home\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[{"title":"Done","type":"check","value":false},{"title":"Created","type":"date","value":""}],"actions":[]}
file "tasks\Home\Fix the tap.json" {"schema":"list.json","Title":"Fix the tap","Done":false,"Created":""}
start task {temp}\tasks
expect focus Your lists, heading level 1
keys down
expect focus Home, list, 1 item
keys right
expect focus Fix the tap

# Goal: Add an item, and its Created field holds the day
keys ctrl+n
expect focus Title of the new item
type Paint the door
keys enter
expect said Added Paint the door
expect focus Paint the door
run findstr /r /c:"Created.: .20[0-9][0-9]-[0-9][0-9]-[0-9][0-9]" "tasks\Home\Paint the door.json"
# Check: An item made before keeps its empty Created
run powershell -NoProfile -Command "if ((Get-Content -Raw 'tasks\Home\Fix the tap.json').Contains('20')) { exit 1 } else { exit 0 }"
# Check: Escape on the new row makes nothing
# TK-4.3b: Ctrl+N opens the box on a row of its own; Escape takes the row away again, says so, and no file is made.
keys ctrl+n
expect focus Title of the new item
type Not this one
keys escape
expect said ! No item made
expect focus Paint the door
run powershell -NoProfile -Command "if (Test-Path 'tasks\Home\Not this one.json') { exit 1 } else { exit 0 }"

# Goal: Give a new item its due date and priority in its title
# TK-4.3c: a date and a !priority at the end of the title fill Due and Priority and leave the title.
file tasks\Work\list.json {"format":"axkit-list 2","details":{"Notes":""},"item fields":[{"title":"Done","type":"check","value":false},{"title":"Due","type":"date","value":""},{"title":"Priority","type":"choice","value":"","options":["High","Normal","Low"]}],"actions":[]}
keys left
expect focus Home, list, 2 items
keys down
expect focus Work, list, 0 items
keys right
keys ctrl+n
expect focus Title of the new item
type Call the plumber tomorrow !high
keys enter
expect said Added Call the plumber, due 
expect said , priority High
expect focus Call the plumber
run findstr /r /c:"Due.: .20[0-9][0-9]-[0-9][0-9]-[0-9][0-9]" "tasks\Work\Call the plumber.json"
run findstr /c:"High" "tasks\Work\Call the plumber.json"
