# Workflow: Select several processes or startup programs and end or remove them together
# axProcess as a navigator (docs/axKit/framework.md FW-4.6, step 7; SPEC PR-3.3a, PR-5.6a; bundle AX-8.16): Space
# selects the process or the startup program you are on, Shift with an arrow selects as it moves, and Delete then
# acts on all the selected after one question that names them.
# SAFETY, processes: the list is this computer's, and the script ends something in it, so it ends only what it
# started itself, under names no other program has: two copies of Windows' own ping, made in the run's folder as
# axkitscript06a.exe and axkitscript06b.exe, each asking this computer itself for 110 echoes (under two minutes,
# nothing leaves the computer). Nothing else is ever selected: the text typed into Find is the copies' own name, the
# step before each selecting key fails unless the row is a copy, the status bar must count two, the question must
# name both, and a script stops at its first failing step.
# SAFETY, startup: nothing here reads or changes what starts with Windows on this computer. In a test run the
# Startup list's five places are a stand-in of the run's own (PR-5.8), seeded, looked at and deleted by
# stand-in.ps1, the same file as 04-startup-turn-off-remove.txt writes, which touches nothing outside it.
file stand-in.ps1 param([string]$runHome, [string]$mode)\n$bytes = [Security.Cryptography.SHA256]::Create().ComputeHash([Text.Encoding]::UTF8.GetBytes($runHome.ToUpperInvariant()))\n$mark = -join ($bytes[0..5] | ForEach-Object { $_.ToString('X2') })\n$root = 'HKCU:\Software\AxKit\Runs\' + $mark\n$user = $root + '\Startup\User\Software\Microsoft\Windows\CurrentVersion'\n$computer = $root + '\Startup\Computer\Software\Microsoft\Windows\CurrentVersion'\nfunction Switch-Byte($name) { $v = (Get-ItemProperty -Path ($user + '\Explorer\StartupApproved\Run') -ErrorAction SilentlyContinue).$name; if ($null -eq $v) { -1 } else { [int]$v[0] } }\nfunction Has($name) { $null -ne (Get-ItemProperty -Path ($user + '\Run') -ErrorAction SilentlyContinue).$name }\n$ok = $true\nif ($mode -eq 'seed') { New-Item -Path ($user + '\Run') -Force | Out-Null; New-Item -Path ($computer + '\Run') -Force | Out-Null; Set-ItemProperty -Path ($user + '\Run') -Name 'Check Alpha' -Value 'C:\Checks\alpha.exe --quiet'; Set-ItemProperty -Path ($user + '\Run') -Name 'Check Beta' -Value 'C:\Checks\beta.exe'; Set-ItemProperty -Path ($computer + '\Run') -Name 'Check Everyone' -Value 'C:\Checks\everyone.exe' }\nif ($mode -eq 'off') { $ok = (Has 'Check Alpha') -and ((Switch-Byte 'Check Alpha') -eq 3) }\nif ($mode -eq 'on') { $ok = (Has 'Check Alpha') -and ((Switch-Byte 'Check Alpha') -eq 2) }\nif ($mode -eq 'removed') { $ok = (-not (Has 'Check Alpha')) -and (Has 'Check Beta') -and ((Switch-Byte 'Check Alpha') -eq -1) }\nif ($mode -eq 'computer') { $ok = ($null -ne (Get-ItemProperty -Path ($computer + '\Run')).'Check Everyone') -and -not (Test-Path ($computer + '\Explorer')) }\nif ($mode -eq 'clean') { Remove-Item -Path $root -Recurse -Force -ErrorAction SilentlyContinue; $runs = 'HKCU:\Software\AxKit\Runs'; foreach ($up in @($runs, (Split-Path $runs))) { $key = Get-Item $up -ErrorAction SilentlyContinue; if ($key -and $key.SubKeyCount -eq 0 -and $key.ValueCount -eq 0) { Remove-Item $up -ErrorAction SilentlyContinue } }; $ok = -not (Test-Path $root) }\nif (-not $ok) { exit 1 }
run powershell -NoProfile -ExecutionPolicy Bypass -File stand-in.ps1 "{temp}\home" seed
file "home\AxKit\startup\yours\Check Morning.lnk" a stand-in for a shortcut
run copy /y %SystemRoot%\System32\PING.EXE axkitscript06a.exe
run copy /y %SystemRoot%\System32\PING.EXE axkitscript06b.exe
run powershell -NoProfile -Command "Start-Process '.\axkitscript06a.exe' -ArgumentList '-n','110','127.0.0.1' -WindowStyle Hidden; Start-Process '.\axkitscript06b.exe' -ArgumentList '-n','110','127.0.0.1' -WindowStyle Hidden"
start process
check commands
# The first read done: the status bar says "Reading..." until then.
expect state "status":["Processes,
expect said Processes,
# By name, so that the two copies stand one after the other.
menu Name
expect said ! Sorted by name

# Check: Delete with one process selected asks about that one, wherever the row you are on is
# Bundle AX-8.16 (the review of 2026-11-27): what a command acts on is the selected rows whenever any is selected,
# one alone too. The first copy selected, the row moved on to the second: the question names the first. Escape ends
# nothing. (Until then the question named the row under the arrow keys.)
do Find in the list
expect state "notice":"Find"
type axkitscript06
keys enter
expect focus "value":"axkitscript06a,
keys space
expect said ! Selected
keys down
expect focus "value":"axkitscript06b,
keys delete
expect state "notice":"End axkitscript06a (process
# A notice holds its answering keys for a moment after it appears (KeyHold).
wait 1200
keys escape
expect state "notice":null
expect list axkitscript06a,
do Deselect all
expect said ! None selected

# Goal: Select two processes with Space and Shift+Down
# Find the first copy, Space selects it, Shift+Down takes the second; the status bar counts them.
do Find in the list
expect state "notice":"Find"
type axkitscript06
keys enter
expect focus "value":"axkitscript06a,
keys space
expect said ! Selected
keys shift+down
expect focus "value":"axkitscript06b,
expect state , 2 selected","

# Check: Closing the question keeps the selected processes
# Delete asks once for both, naming them; Escape keeps both and the selection.
keys delete
expect state "notice":"End 2 processes?"
expect state Anything unsaved in them is lost.
expect state axkitscript06a (process
expect state axkitscript06b (process
# A notice holds its answering keys for a moment after it appears (KeyHold).
wait 1200
keys escape
expect state "notice":null
expect list axkitscript06a,
expect state , 2 selected","

# Goal: End the selected processes together
# The question again, Enter: both ended, one sentence, the selection given up, the list read again without them.
keys delete
expect state "notice":"End 2 processes?"
expect state axkitscript06a (process
expect state axkitscript06b (process
wait 1200
keys enter
expect said ! Ended 2 processes
refuse list axkitscript06
# Windows itself has neither any more (a read by name, of the copies' own name alone).
run powershell -NoProfile -Command "if (Get-Process -Name axkitscript06a, axkitscript06b -ErrorAction SilentlyContinue) { exit 1 }"

# Goal: Select two startup programs and remove them together
# Alt+3: the stand-in's four entries. Space on the first, Shift+Down to the second; Delete asks once, Cancel the
# default; Remove takes both out, one sentence, and the list is read again with the two that stay.
keys alt+3
expect said ! Startup, 4. Check Alpha, enabled, current user's Run key
keys space
expect said ! Selected
keys shift+down
expect focus Check Beta, enabled, current user's Run key, selected
expect state , 2 selected","
keys delete
expect state "notice":"Remove 2 startup programs from the programs that start with Windows?"
expect state Check Alpha, Check Beta
wait 1200
# Check: Enter on the question removes nothing
# Cancel is the default answer of a removal: Enter keeps both.
keys enter
expect state "notice":null
expect list Check Alpha, enabled
expect list Check Beta, enabled
keys delete
expect state "notice":"Remove 2 startup programs from the programs that start with Windows?"
wait 1200
keys alt+r
expect said ! Removed 2 startup programs
expect focus Startup, 2
refuse list Check Alpha
refuse list Check Beta
expect list Check Everyone, enabled, computer's Run key
expect list Check Morning, enabled, your Startup folder

# Check: The stand-in's registry key is deleted at the end
# The stand-in's key deleted: nothing of the run is left in the registry.
run powershell -NoProfile -ExecutionPolicy Bypass -File stand-in.ps1 "{temp}\home" clean
