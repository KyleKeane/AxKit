# Workflow: Change the colours with the theme file
# axDown: the theme items every window has (docs/axKit/theme.md TH-4, TH-5.3). The theme file lives in the run's own
# home, so nothing of the user's is touched: Reload with no file says what makes one; Edit theme file writes the
# template and opens it in axJSON; Reload then says so; Theme on toggles the file's switch and says which.
file notes.md # Notes\n\nA line.
start down {temp}\notes.md
expect focus notes.md
# Check: Reloading with no theme file says how to make one
menu Reload theme
expect said ! No theme file: Options, Theme, Edit theme file makes one
# Goal: Make a theme file and open it
menu Edit theme file
adopt
expect state , axJSON, 
expect state \theme.json","notice"
expect text axkit-theme 1
close
use 1
# Goal: Reload the theme
menu Reload theme
expect said ! Theme reloaded
# Goal: Turn the theme off and on
menu Theme on
expect said ! Theme on, not checked
# Check: Theme on turns the theme back on
menu Theme on
expect said ! Theme on, checked
# Check: Word wrap is kept in the theme file
# TH-6: F11 writes the editors' word wrap into the same file; axJSON on it shows the value; back on for the scripts after.
do Word wrap
expect said ! Word wrap, not checked
menu Edit theme file
adopt
expect state , axJSON, 
expect state \theme.json","notice"
expect text "wordWrap": false
close
use 1
do Word wrap
expect said ! Word wrap, checked
