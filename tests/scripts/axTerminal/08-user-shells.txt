# Workflow: Add a shell of your own
# axTerminal: the user's own shells (AT-1.6a), section 3 of docs/axTerminal/nvda-test-plan.md, the steps on Add a
# shell and Remove a shell. Switch shell needs menu mode, which only the foreground window enters: "menu" runs the
# same items by their palette names ("1. Test cmd (cmd.exe /k)" to switch, "1. Remove Test cmd" to remove).
mkdir work\myproject
file home\AxKit\axTerminal.json {"checkForUpdates": false, "shell": "powershell"}
# The reminders agreed to already (AX-11.5): a user's shell starts as the custom shell does, under its consent.
file home\AxKit\consent.json {"items":[{"app":"axTerminal","target":"Windows PowerShell","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Custom shell","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"PowerShell 7","version":"0","when":"2026-10-31T00:00:00"}]}
start terminal {temp}\work\myproject
expect said Ready

# Goal: Add a shell of your own
# Add a shell: one form (bundle AX-8.10), the name then the command line, the focus in the name and the explanation
# said once; a blank or taken name, or a blank command line, is said with the focus in that field, the form kept.
# With none yet, Remove a shell holds one line that says so; it does nothing, so it is no command of the list
# (docs/axKit/app-kit.md AK-2.5: the line that says a list is empty is never an entry).
refuse commands No shells of your own yet
menu Add a shell
expect state "notice":"Add a shell"
expect focus "name":"Type the shell's name"
expect said A shell of your own: the name Switch shell and the New session notice show for it
keys enter
expect said ♪ Edge
expect said Type a name
expect state "notice":"Add a shell"
expect focus "name":"Type the shell's name"
type Command Prompt
keys enter
expect said A shell named Command Prompt is already there
expect focus "name":"Type the shell's name"
keys ctrl+a
type Test cmd
# Enter in a one-line field presses Add: the command line is still empty, so the focus goes there.
keys enter
expect said Type the command line
expect state "notice":"Add a shell"
expect focus "name":"Type the command line that starts it"
type cmd.exe /k
# Tab reaches the buttons after the two fields; Shift+Tab comes back; Ctrl+Enter adds from the field.
keys tab
expect focus Add
keys tab
expect focus Cancel
keys shift+tab shift+tab
expect focus "name":"Type the command line that starts it"
keys ctrl+enter
expect said Added Test cmd
expect state "notice":null
refuse said New session

# Check: A new shell is listed in Switch shell and in Remove a shell
# Listed in Switch shell after the built-in shells, and in Remove a shell.
expect commands 1. Test cmd (cmd.exe /k), Session, Switch shell
expect commands 1. Remove Test cmd, Session, Remove a shell

# Goal: Switch to a shell of your own
# Started from Switch shell: the title and the system line name it; cmd's own output in the terminal.
menu 1. Test cmd
expect said Test cmd
expect state "myproject - Test cmd - axTerminal, access layer for your shell"
expect commands 1. Test cmd (cmd.exe /k), Session, Switch shell, checked
do Terminal
expect text "System: Switched to Test cmd in
do Command field
type echo hello-user-shell
keys enter
do Terminal
expect text hello-user-shell
do Command field

# Check: The New session notice lists the shell of your own, selected
# The New session notice lists it after the built-in shells, selected while it is the shell in use.
do New session
expect state "notice":"New session"
keys tab
expect focus folder..."}
keys tab
expect focus "role":"RadioButton"
expect focus "value":"Test cmd"}
keys escape
expect state "notice":null
close

# Check: The shell in use starts again in the next window
# Remembered for the next window: the same shell starts again.
start terminal {temp}\work\myproject
expect state "myproject - Test cmd - axTerminal, access layer for your shell"
do Terminal
expect text "System: Shell started: Test cmd in
do Command field
