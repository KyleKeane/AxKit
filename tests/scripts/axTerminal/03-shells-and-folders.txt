# Workflow: Switch shells and folders
# axTerminal, section 3 of docs/axTerminal/nvda-test-plan.md: shells and folders (AT-1.2, AT-1.6, AT-1.7). Alt+S, W
# (the Switch shell submenu) needs menu mode, which only the foreground window enters: "menu" runs the same items.
mkdir work\myproject
# The run's own settings: no update check, which would go to the network and log an ERROR when it fails.
file home\AxKit\axTerminal.json {"checkForUpdates": false}
# The access layers' reminders agreed to already (AX-11.5): this script is about what comes after them.
file home\AxKit\consent.json {"items":[{"app":"axClaude","target":"Claude Code","version":"0","when":"2026-10-31T00:00:00"},{"app":"axGIT","target":"Git","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Windows PowerShell","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"PowerShell 7","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Command Prompt","version":"0","when":"2026-10-31T00:00:00"},{"app":"axTerminal","target":"Git Bash","version":"0","when":"2026-10-31T00:00:00"},{"app":"axPass","target":"KeePassXC","version":"0","when":"2026-10-31T00:00:00"}]}
start terminal {temp}\work\myproject
expect said ♪ Ready

# Goal: Start a new session in another shell
# Step 1: Ctrl+N, the New session notice: its text and controls; Command Prompt chosen and started.
do New session
expect state "notice":"New session"
expect state Choose the shell and the folder.
expect focus "name":"New session"
expect focus "value":"Folder:
keys tab
expect focus folder..."}
keys tab
expect focus "role":"RadioButton"
expect focus "value":"Windows PowerShell"}
keys tab
expect focus Start session"}
keys tab
expect focus "value":"Cancel"}
keys tab
expect focus "name":"New session"
keys tab
expect focus folder..."}
keys tab
expect focus "role":"RadioButton"
keys down
expect focus "value":"Command Prompt"}
keys enter
expect said New session in myproject
expect said Ready
expect state "title":"axTerminal, access layer for your shell folder 
expect state Command Prompt","notice"
do Terminal
expect focus "name":"Terminal"
expect text "System: New session: Command Prompt in
do Command field
type echo hello
keys enter
expect said Done
expect said ♪ Done
# Check: Command Prompt says Done, never Failed
# Command Prompt gives no exit code: "Done", never "Failed".
refuse said Failed
# Check: A program in the current folder runs only by its path
# A program in the current folder is run as .\name since 2026-11-09 (NoDefaultCurrentDirectoryInExePath, security audit
# S-1): the bare name is not found, .\hello runs it.
file work\myproject\hello.cmd @echo planted
type hello
keys enter
expect said Done
do Terminal
expect text not recognized
do Command field
type .\hello
keys enter
expect said Done
do Terminal
expect text planted
do Command field
# Goal: Switch to another shell
# The Switch shell submenu with Command Prompt checked (the palette's list carries the check), then Windows PowerShell.
expect commands Command Prompt, Session, Switch shell, checked
menu Windows PowerShell
expect said Windows PowerShell
expect said Ready
do Terminal
expect text "System: Switched to Windows PowerShell in
expect text "echo hello"
do Command field

# Check: Git Bash shows its prompt, its exit codes and its folder in the title
# Step 2: Git Bash: the status bar shows $, the title names the folder and Git Bash; ls; a failing command; cd ..
# follows in the title; Ctrl+Q; back to Windows PowerShell. Fails at once where Git for Windows is not installed.
menu Git Bash
expect said Git Bash
expect said Ready
expect state "title":"axTerminal, access layer for your shell folder 
expect state \myproject Git Bash","notice"
# The prompt in the status bar is the user's PS1 with its rows joined (user, host, MINGW64, the path), ending in $.
expect state "status":["Idle",
expect state $"]}
type ls
keys enter
expect said Done
expect said ♪ Done
type (exit 3)
keys enter
expect said Failed, exit code 3
expect said ♪ Done
do What is running
expect said Idle, Git Bash,
type cd ..
keys enter
expect said Done
expect said ♪ Done
expect state "title":"axTerminal, access layer for your shell folder 
expect state \work Git Bash","notice"
type cd myproject
keys enter
expect said Done
expect said ♪ Done
menu Windows PowerShell
expect said Windows PowerShell
expect said Ready
expect state "title":"axTerminal, access layer for your shell folder 
expect state \myproject Windows PowerShell","notice"

# Goal: Hear the current folder's full path
# Step 3: cd .. in PowerShell: the title and the status bar name the parent folder; Ctrl+P says its full path.
type cd ..
keys enter
expect said Done
expect said ♪ Done
expect state "title":"axTerminal, access layer for your shell folder 
expect state \work Windows PowerShell","notice"
expect state \\work\u003E"]
do Say current folder
expect said {temp}\work
# Ctrl+W (File Explorer) and Alt+S, I, C (axClaude on the folder) stay manual: both start a program of their own
# outside the test desktop's control, and the Alt keys need menu mode.
type cd myproject
keys enter
expect said Done
expect said ♪ Done

# Goal: Restart the shell
# Step 4: Ctrl+Shift+R while idle restarts at once, keeping the transcript; while a command runs, a notice asks first.
do Restart shell
expect said Restarting
expect said Ready
do Terminal
# The view's text answer holds the last 60 lines: the transcript kept is checked by the last command before.
expect text "cd myproject"
expect text "System: Restarting the shell","System: Shell started:
do Command field
type ping -n 30 127.0.0.1
keys enter
expect said Sent
expect state Running: ping
do Restart shell
expect state "notice":"Restart the shell?"
keys escape
expect focus "name":"Command"
expect state "notice":null
expect state Running: ping
do Restart shell
expect state "notice":"Restart the shell?"
keys enter
expect said Restarting
expect said Ready
expect state "status":["Idle"

# Step 5: Session, Open recent folder stays manual: the submenu is filled as it opens, so the palette (and "menu")
# does not list its folders, and opening it needs menu mode.

# Goal: Start the shell again after it ended
# Step 6: exit ends the shell; Enter in the field says so; Ctrl+Shift+R starts it again.
type exit
keys enter
expect said The shell ended
expect state "status":["Shell ended"
do Terminal
expect text "System: The shell ended with code 0"
do Command field
type echo kept
keys enter
expect said The shell ended. Ctrl+Shift+R starts it again
expect focus "value":"echo kept"}
do Restart shell
