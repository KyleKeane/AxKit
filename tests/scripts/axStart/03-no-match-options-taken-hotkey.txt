# Workflow: Hear that nothing matches, turn the spoken match off and meet a hotkey that is taken
# axStart, docs/axStart/nvda-test-plan.md, what 01-basics and 02-typing-and-the-hotkey leave: 2.5 (nothing matches,
# said while typing, and again after Backspace), 3.6 (Ctrl+I on a Store app), 5.3 (Say the first match while typing
# off: typing says nothing, Ctrl+I says the match; on again), 4.2 (Shift with a letter refused) and 4.5 (a key another
# program holds: said, and the key in use kept). SPEC AS-3.2, AS-4. The key that is taken is one nobody uses, held for
# half a minute by a hidden Windows PowerShell the script starts itself (hold.ps1), apart from the runner's command
# (run's own children end with the command); the hotkey of the run is unlike the other two scripts'.
# Not run: 2.6's name with an accent (it needs such a program in the user's Start menu), starting programs, File
# Explorer, the Start menu and signing in (outside the window).
file home\AxKit\axStart.json {"checkForUpdates": false, "hotkey": "Ctrl+Alt+Shift+F8", "sayFirstMatch": true}
file hold.ps1 Add-Type -Namespace N -Name H -MemberDefinition '[DllImport("user32.dll")] public static extern bool RegisterHotKey(System.IntPtr h, int id, uint m, uint k);'\n[N.H]::RegisterHotKey([IntPtr]::Zero, 1, 7, 0x76) | Out-File held.txt\nStart-Sleep -Seconds 30\n
run powershell -NoProfile -Command "Start-Process powershell -ArgumentList '-NoProfile','-ExecutionPolicy','Bypass','-File','hold.ps1' -WindowStyle Hidden"

start start
expect focus Search programs
# The programs are read once at the start; the hidden PowerShell has its key by then.
wait 5000

# 2.5 A text nothing matches is said after the pause, with the text; Backspace: the new text, said again.
type zzqqxx
expect said ! No programs match zzqqxx
keys backspace
expect focus "value":"zzqqx"
expect said ! No programs match zzqqx

# 3.6 Ctrl+I on a Store app: its name and its kind.
keys ctrl+a
type calculator
expect said ! Calculator, 1 of
keys ctrl+i
expect said ! Calculator, Store app

# 5.3 Say the first match while typing off: typing says nothing, Ctrl+I says the match all the same; on again.
menu Say the first match while typing
expect said ! Say the first match while typing, not checked
keys ctrl+a
type calculato
expect focus "value":"calculato"
expect quiet 2000
keys ctrl+i
expect said ! Calculator, Store app
menu Say the first match while typing
expect said ! Say the first match while typing, checked
keys backspace
expect said ! Calculator, 1 of

# 4.2 The Hotkey notice names the key in use; Shift with a letter is refused like a letter alone.
menu Hotkey
expect state "notice":"Hotkey"
expect state Now: Ctrl+Alt+Shift+F8.
expect focus "name":"Press the new key","role":"Text","value":"Ctrl+Alt+Shift+F8"
keys shift+a
expect said ! Use Ctrl, Alt or the Windows key with it
expect focus "value":"Ctrl+Alt+Shift+F8"

# 4.5 A key another program holds: said on Enter, the field back with its text, the key in use unchanged.
keys ctrl+alt+shift+f7
expect focus "value":"Ctrl+Alt+Shift+F7"
keys enter
expect said ! Ctrl+Alt+Shift+F7 is taken by another program
expect state "notice":null
expect focus "name":"Search programs","role":"Text","value":"calculat"
menu Hotkey
expect state Now: Ctrl+Alt+Shift+F8.
expect focus "value":"Ctrl+Alt+Shift+F8"
keys escape
expect state "notice":null

# Quit axStart, so that its hotkey goes with it.
menu Quit axStart
expect closed
