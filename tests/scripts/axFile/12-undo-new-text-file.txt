# Workflow: Take back a new text file
# axFile, docs/axFile/user-guide.md "File operations", Ctrl+Z (SPEC AF-4.14): a new text file is removed by Ctrl+Z,
# as a new folder is (03-operations.txt), and the list is as it was.
mkdir u
file u\keep.txt keep
start file {temp}\u
expect focus keep.txt

# Goal: Undo a new text file
# Ctrl+T makes "New file.txt" with its rename box; Escape keeps the name; Ctrl+Z removes the file.
do New text file
expect focus "value":"New file.txt"
keys escape
expect focus "value":"New file.txt"
expect list "count":2
keys ctrl+z
expect said ! Undone: New file.txt removed
expect list "count":1
run cmd /c if exist "u\New file.txt" (exit 1) else (exit 0)

# Goal: Make a new Markdown file and a new JSON file
# Actions, New Markdown file and New JSON file (AF-4.1): the same rename box on "New file.md" and "New file.json".
do New Markdown file
expect focus "value":"New file.md"
keys escape
# Goal: Make a new JSON file
do New JSON file
expect focus "value":"New file.json"
keys escape
expect list "count":3
# Check: Undo removes each new file, the newest first
keys ctrl+z
expect said ! Undone: New file.json removed
keys ctrl+z
expect said ! Undone: New file.md removed
expect list "count":1
