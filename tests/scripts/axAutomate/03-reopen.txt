# Workflow: Close axAutomate and come back to the folder you had
# axAutomate: started with no folder, the folder used last opens (docs/axAutomate/nvda-test-plan.md 4.2; SPEC.md
# AU-4.1, AU-6a.1). The folder used last is the one a start named or Ctrl+O chose (Ctrl+O's alone until 2026-11-14).
# Ctrl+O itself, step 4.1, is Windows' folder picker, which the channel cannot drive: the script names the folder at
# its first start, and checks that a start without a folder opens it, twice.
mkdir flows
file flows\02-two.txt # A workflow.\n\nstart down\nwait 100\nclose
file home\AxKit\axAutomate.json {"checkForUpdates": false}

# AU-4.1 Started on a folder: it opens, and is the folder used last from then on.
start automate {temp}\flows
expect state "title":"flows - axAutomate"
expect focus 02-two, workflow, 3 steps
close
expect closed

# 4.2 Started with no folder: the folder used last, at its first row.
start automate
expect state "title":"flows - axAutomate"
expect focus 02-two, workflow, 3 steps
# Inside the workflow when the window closes (Alt+F4): the next start is at the folder again, not in the workflow.
keys right
expect said 02-two, 3 steps, 1. Start down
close
expect closed
start automate
expect state "title":"flows - axAutomate"
expect focus 02-two, workflow, 3 steps
